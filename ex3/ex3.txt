#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <string.h>
#include <sys/types.h>
#include <sys/wait.h>

#define MAX_LINE 1024
#define MAX_ARGS 64

int main(int argc, char *argv[], char *envp[]) {
    char line[MAX_LINE];
    char *args[MAX_ARGS];

    while (1) {
        printf("myshell> ");
        fflush(stdout);

        if (fgets(line, sizeof(line), stdin) == NULL) {
            break;
        }

        line[strcspn(line, "\n")] = 0;

        if (strlen(line) == 0) {
            continue;
        }

        if (strcmp(line, "exit") == 0) {
            break;
        }

        int i = 0;
        char *token = strtok(line, " ");
        while (token != NULL && i < MAX_ARGS - 1) {
            args[i++] = token;
            token = strtok(NULL, " ");
        }
        args[i] = NULL;

        if (args[0] == NULL) {
            continue;
        }

        pid_t pid = fork();

        if (pid < 0) {
            perror("fork failed");
            exit(1);
        } 
        else if (pid == 0) {
            char path[256];
            if (args[0][0] == '/') {
                execve(args[0], args, envp);
            } else {
                snprintf(path, sizeof(path), "/bin/%s", args[0]);
                execve(path, args, envp);

                snprintf(path, sizeof(path), "/usr/bin/%s", args[0]);
                execve(path, args, envp);
            }

            perror("execve failed");
            exit(1);
        } 
        else {
            printf("[Background process started with PID: %d]\n", pid);
        }
    }

    return 0;
}
